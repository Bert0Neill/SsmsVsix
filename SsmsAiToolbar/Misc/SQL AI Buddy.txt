record OllamaRequest(
    string model,
    string prompt,
    bool stream,
    OllamaOptions options
);

record OllamaOptions(
    int num_predict,
    double temperature,
    double top_p,
    string stop
);


using System.Net.Http;
using System.Text;
using System.Text.Json;

var schema = """
Tables:
Orders(Id, CustomerId, OrderDate, Total)
Customers(Id, Name, CountryId)
Countries(Id, Name)
""";

var userSql = "SELECT o.Id, o.Total FROM Orders o JOIN Customers c ON";

var prompt = $"""
You are a SQL autocomplete engine.

Rules:
- Return ONLY suggestions.
- No explanations.
- One suggestion per line.
- Use only provided schema.
- Valid T-SQL only.

Database schema:
{schema}

User SQL (cursor at end):
{userSql}
""";

var req = new OllamaRequest(
    model: "qwen2.5-coder:7b",   // good for SQL
    prompt: prompt,
    stream: false,
    options: new OllamaOptions(
        num_predict: 64,
        temperature: 0.1,
        top_p: 0.9,
        stop: "\n\n"
    )
);

using var http = new HttpClient();
var res = await http.PostAsync(
    "http://localhost:11434/api/generate",
    new StringContent(JsonSerializer.Serialize(req), Encoding.UTF8, "application/json")
);

var json = await res.Content.ReadAsStringAsync();
using var doc = JsonDocument.Parse(json);

var suggestions = doc.RootElement.GetProperty("response").GetString();
Console.WriteLine(suggestions);

-------------------Streaming-------------------

var req = new
{
    model = "qwen2.5-coder:7b",
    prompt,
    stream = true,
    options = new { temperature = 0.1, num_predict = 64 }
};

using var res = await http.PostAsync(
    "http://localhost:11434/api/generate",
    new StringContent(JsonSerializer.Serialize(req), Encoding.UTF8, "application/json")
);

using var stream = await res.Content.ReadAsStreamAsync();
using var reader = new StreamReader(stream);

while (!reader.EndOfStream)
{
    var line = await reader.ReadLineAsync();
    if (string.IsNullOrWhiteSpace(line)) continue;

    using var doc = JsonDocument.Parse(line);
    if (doc.RootElement.TryGetProperty("response", out var r))
        Console.Write(r.GetString());
}

static IReadOnlyDictionary<string, (string Schema, string Table)> ExtractTables(string sql)
{
    var result = new Dictionary<string, (string, string)>(StringComparer.OrdinalIgnoreCase);

    var regex = new Regex(
        @"(?:FROM|JOIN)\s+(?<table>(?:\[[^\]]+\]|\w+)(?:\.(?:\[[^\]]+\]|\w+))?)\s*(?:AS\s+)?(?<alias>\w+)?",
        RegexOptions.IgnoreCase | RegexOptions.Multiline);

    foreach (Match m in regex.Matches(sql))
    {
        var table = m.Groups["table"].Value.Replace("[", "").Replace("]", "");
        var alias = m.Groups["alias"].Success ? m.Groups["alias"].Value : null;

        var parts = table.Split('.');
        var schema = parts.Length == 2 ? parts[0] : "dbo";
        var name = parts.Length == 2 ? parts[1] : parts[0];

        result[alias ?? name] = (schema, name);
    }

    return result;
}

SELECT
    t.name AS TableName,
    s.name AS SchemaName,
    c.name AS ColumnName,
    ty.name AS DataType,
    c.is_nullable
FROM sys.tables t
JOIN sys.schemas s ON t.schema_id = s.schema_id
JOIN sys.columns c ON t.object_id = c.object_id
JOIN sys.types ty ON c.user_type_id = ty.user_type_id
WHERE s.name = @Schema AND t.name = @Table
ORDER BY c.column_id;


------------------Primary Keys---------------------------
SELECT
    kc.name AS PKName,
    c.name AS ColumnName
FROM sys.key_constraints kc
JOIN sys.index_columns ic ON kc.parent_object_id = ic.object_id AND kc.unique_index_id = ic.index_id
JOIN sys.columns c ON ic.object_id = c.object_id AND ic.column_id = c.column_id
WHERE kc.type = 'PK'
AND OBJECT_SCHEMA_NAME(kc.parent_object_id) = @Schema
AND OBJECT_NAME(kc.parent_object_id) = @Table;

--------------Foreign keys-----------------------
SELECT
    fk.name AS FKName,
    cp.name AS ColumnName,
    rt.name AS RefTable,
    cr.name AS RefColumn
FROM sys.foreign_keys fk
JOIN sys.foreign_key_columns fkc ON fk.object_id = fkc.constraint_object_id
JOIN sys.columns cp ON fkc.parent_object_id = cp.object_id AND fkc.parent_column_id = cp.column_id
JOIN sys.tables rt ON fkc.referenced_object_id = rt.object_id
JOIN sys.columns cr ON fkc.referenced_object_id = cr.object_id AND fkc.referenced_column_id = cr.column_id
WHERE OBJECT_SCHEMA_NAME(fk.parent_object_id) = @Schema
AND OBJECT_NAME(fk.parent_object_id) = @Table;

------------------Indexes (non pk)---------------------
SELECT
    i.name AS IndexName,
    c.name AS ColumnName,
    i.is_unique
FROM sys.indexes i
JOIN sys.index_columns ic ON i.object_id = ic.object_id AND i.index_id = ic.index_id
JOIN sys.columns c ON ic.object_id = c.object_id AND ic.column_id = c.column_id
WHERE i.is_primary_key = 0
AND OBJECT_SCHEMA_NAME(i.object_id) = @Schema
AND OBJECT_NAME(i.object_id) = @Table;

-------------------Check \ contraints--------------------
SELECT
    dc.name AS ConstraintName,
    c.name AS ColumnName,
    dc.definition
FROM sys.default_constraints dc
JOIN sys.columns c ON dc.parent_object_id = c.object_id AND dc.parent_column_id = c.column_id
WHERE OBJECT_SCHEMA_NAME(dc.parent_object_id) = @Schema
AND OBJECT_NAME(dc.parent_object_id) = @Table;
