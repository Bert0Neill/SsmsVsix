static IReadOnlyDictionary<string, (string Schema, string Table)> ExtractTables(string sql)
{
    var result = new Dictionary<string, (string, string)>(StringComparer.OrdinalIgnoreCase);

    var regex = new Regex(
        @"(?:FROM|JOIN)\s+(?<table>(?:\[[^\]]+\]|\w+)(?:\.(?:\[[^\]]+\]|\w+))?)\s*(?:AS\s+)?(?<alias>\w+)?",
        RegexOptions.IgnoreCase | RegexOptions.Multiline);

    foreach (Match m in regex.Matches(sql))
    {
        var table = m.Groups["table"].Value.Replace("[", "").Replace("]", "");
        var alias = m.Groups["alias"].Success ? m.Groups["alias"].Value : null;

        var parts = table.Split('.');
        var schema = parts.Length == 2 ? parts[0] : "dbo";
        var name = parts.Length == 2 ? parts[1] : parts[0];

        result[alias ?? name] = (schema, name);
    }

    return result;
}

SELECT
    t.name AS TableName,
    s.name AS SchemaName,
    c.name AS ColumnName,
    ty.name AS DataType,
    c.is_nullable
FROM sys.tables t
JOIN sys.schemas s ON t.schema_id = s.schema_id
JOIN sys.columns c ON t.object_id = c.object_id
JOIN sys.types ty ON c.user_type_id = ty.user_type_id
WHERE s.name = @Schema AND t.name = @Table
ORDER BY c.column_id;

using EnvDTE;
using Microsoft.VisualStudio.Shell;

public sealed class EditorTracker
{
    private readonly DTE _dte;
    private DocumentEvents _docEvents;

    public EditorTracker(DTE dte)
    {
        _dte = dte;
        HookEvents();
    }

    private void HookEvents()
    {
        _docEvents = _dte.Events.DocumentEvents;
        _docEvents.DocumentActivated += OnDocumentActivated;
    }

    private void OnDocumentActivated(Document doc)
    {
        if (!IsSqlDocument(doc))
            return;

        // User clicked into a SQL editor
        HandleEditorActivated(doc);
    }

    private static bool IsSqlDocument(Document doc)
        => doc.Language == "SQL";
}

SchemaCache LoadSchema(IConnection connection)
{
    var serverConn = new ServerConnection
    {
        ServerInstance = connection.ServerName,
        LoginSecure = true // or SQL auth if needed
    };

    var server = new Server(serverConn);
    var db = server.Databases[connection.DatabaseName];

    var schema = new SchemaCache();

    foreach (Table table in db.Tables)
    {
        if (table.IsSystemObject)
            continue;

        var t = new TableSchema
        {
            Name = table.Name,
            Schema = table.Schema
        };

        // Columns
        foreach (Column col in table.Columns)
        {
            t.Columns.Add(new ColumnSchema
            {
                Name = col.Name,
                Type = col.DataType.Name,
                IsNullable = col.Nullable
            });
        }

        // PK
        foreach (Index idx in table.Indexes)
        {
            if (!idx.IsPrimaryKey) continue;

            t.PrimaryKey = idx.IndexedColumns
                .Cast<IndexedColumn>()
                .Select(c => c.Name)
                .ToList();
        }

        // FKs
        foreach (ForeignKey fk in table.ForeignKeys)
        {
            t.ForeignKeys.Add(new ForeignKeySchema
            {
                Name = fk.Name,
                ReferencedTable = fk.ReferencedTable,
                Columns = fk.Columns
                    .Cast<ForeignKeyColumn>()
                    .Select(c => (c.Name, c.ReferencedColumn))
                    .ToList()
            });
        }

        schema.Tables.Add(t);
    }

    return schema;
}

Dictionary<string, List<JoinEdge>> BuildJoinGraph(SchemaCache schema)
{
    var graph = new Dictionary<string, List<JoinEdge>>();

    foreach (var table in schema.Tables)
    {
        foreach (var fk in table.ForeignKeys)
        {
            graph.TryAdd(table.Name, new());
            graph[table.Name].Add(new JoinEdge
            {
                FromTable = table.Name,
                ToTable = fk.ReferencedTable,
                Columns = fk.Columns
            });
        }
    }

    return graph;
}
