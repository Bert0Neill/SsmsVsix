WITH SalesData AS
(
    SELECT
        soh.SalesOrderID,
        soh.OrderDate,
        soh.CustomerID,
        soh.SalesPersonID,
        soh.TerritoryID,
        soh.TotalDue,

        c.AccountNumber,
        p.FirstName + ' ' + p.LastName AS CustomerName,

        sp.BusinessEntityID AS SalesRepID,
        pp.FirstName + ' ' + pp.LastName AS SalesRepName,

        st.Name AS TerritoryName,
        st.CountryRegionCode,

        sod.ProductID,
        sod.OrderQty,
        sod.LineTotal,

        pr.Name AS ProductName,
        pc.Name AS CategoryName,
        psc.Name AS SubCategoryName

    FROM Sales.SalesOrderHeader soh

    INNER JOIN Sales.SalesOrderDetail sod
        ON soh.SalesOrderID = sod.SalesOrderID

    INNER JOIN Sales.Customer c
        ON soh.CustomerID = c.CustomerID

    LEFT JOIN Person.Person p
        ON c.PersonID = p.BusinessEntityID

    LEFT JOIN Sales.SalesPerson sp
        ON soh.SalesPersonID = sp.BusinessEntityID

    LEFT JOIN Person.Person pp
        ON sp.BusinessEntityID = pp.BusinessEntityID

    LEFT JOIN Sales.SalesTerritory st
        ON soh.TerritoryID = st.TerritoryID

    INNER JOIN Production.Product pr
        ON sod.ProductID = pr.ProductID

    LEFT JOIN Production.ProductSubcategory psc
        ON pr.ProductSubcategoryID = psc.ProductSubcategoryID

    LEFT JOIN Production.ProductCategory pc
        ON psc.ProductCategoryID = pc.ProductCategoryID

    WHERE soh.OrderDate >= '2013-01-01'
),

CustomerTotals AS
(
    SELECT
        CustomerID,
        CustomerName,
        TerritoryName,

        COUNT(DISTINCT SalesOrderID) AS TotalOrders,
        SUM(LineTotal) AS TotalSales,
        AVG(LineTotal) AS AvgLineValue,

        RANK() OVER
        (
            PARTITION BY TerritoryName
            ORDER BY SUM(LineTotal) DESC
        ) AS CustomerRank

    FROM SalesData
    GROUP BY
        CustomerID,
        CustomerName,
        TerritoryName
),

CategorySales AS
(
    SELECT
        CategoryName,

        SUM(LineTotal) AS CategoryRevenue,

        DENSE_RANK() OVER
        (
            ORDER BY SUM(LineTotal) DESC
        ) AS RevenueRank

    FROM SalesData
    GROUP BY CategoryName
)

SELECT
    sd.TerritoryName,
    sd.CountryRegionCode,

    ct.CustomerName,
    ct.TotalOrders,
    ct.TotalSales,
    ct.AvgLineValue,
    ct.CustomerRank,

    cs.CategoryName,
    cs.CategoryRevenue,
    cs.RevenueRank,

    sd.SalesRepName,

    CASE
        WHEN ct.TotalSales > 100000 THEN 'VIP'
        WHEN ct.TotalSales > 50000 THEN 'Preferred'
        ELSE 'Standard'
    END AS CustomerTier,

    MIN(sd.OrderDate) AS FirstOrderDate,
    MAX(sd.OrderDate) AS LastOrderDate,

    COUNT(DISTINCT sd.ProductID) AS UniqueProductsPurchased

FROM SalesData sd

INNER JOIN CustomerTotals ct
    ON sd.CustomerID = ct.CustomerID

INNER JOIN CategorySales cs
    ON sd.CategoryName = cs.CategoryName

GROUP BY
    sd.TerritoryName,
    sd.CountryRegionCode,

    ct.CustomerName,
    ct.TotalOrders,
    ct.TotalSales,
    ct.AvgLineValue,
    ct.CustomerRank,

    cs.CategoryName,
    cs.CategoryRevenue,
    cs.RevenueRank,

    sd.SalesRepName

HAVING SUM(sd.LineTotal) > 10000

ORDER BY
    ct.TotalSales DESC,
    cs.RevenueRank ASC;
